    20→function rnd<T>(arr: T[]): T {
    21→  return arr[Math.floor(Math.random() * arr.length)]
    22→}
    23→function randInt(min: number, max: number): number {
    24→  return min + Math.floor(Math.random() * (max - min + 1))
    25→}
    26→function weightedCatalogItem() {
    27→  const total = CATALOG.reduce((s, i) => s + (i.weight ?? 1), 0)
    28→  let r = Math.random() * total
    29→  for (const item of CATALOG) {
    30→    r -= item.weight ?? 1
    31→    if (r <= 0) return item
    32→  }
    33→  return CATALOG[0]
    34→}

    36→export async function getCategoryMult(category: string): Promise<number> {
    37→  const idx = await db.marketIndex.findUnique({ where: { category } })
    38→  const events = await db.marketEvent.findMany({
    39→    where: { category, expiresAt: { gt: new Date() } },
    40→  })
    41→  const eventMult = events.reduce((m, e) => m * (1 + e.magnitude), 1)
    42→  return (idx?.multiplier ?? 1) * eventMult
    43→}
    44→
    45→function conditionWeighted(): string {
    46→  const r = Math.random()
    47→  if (r < 0.08) return 'new'
    48→  if (r < 0.35) return 'excellent'
    49→  if (r < 0.7) return 'good'
    50→  if (r < 0.93) return 'used'
    51→  return 'parts'
    52→}

    68→async function marketTick(tick: number) {
    69→  const cats = await db.marketIndex.findMany()
    70→  if (!cats.length) return
    71→  // случайное блуждание 2 категорий
    72→  for (let i = 0; i < 2; i++) {
    73→    const c = rnd(cats)
    74→    const delta = (Math.random() - 0.5) * 0.04
    75→    const mult = Math.min(1.75, Math.max(0.55, c.multiplier + delta))
    76→    await db.marketIndex.update({ where: { category: c.category }, data: { multiplier: mult } })
    77→  }
    78→  // небольшие события каждые ~3 минуты
    79→  if (tick % 12 === 0 && Math.random() < 0.35) {
    80→    const c = rnd(cats)
    81→    const up = Math.random() < 0.5
    82→    const magnitude = 0.05 + Math.random() * 0.12
    83→    const templates = up
    84→      ? [
    85→          `Спрос на «${catLabel(c.category)}» вырос: покупатели смели лучшие предложения`,
    86→          `Аналитики: категория «${catLabel(c.category)}» недооценена, толпа ринулась скупать`,
    87→        ]
    88→      : [
    89→          `На рынок выбросили партию товаров категории «${catLabel(c.category)}» — цены поехали вниз`,
    90→          `Сезонный спад: «${catLabel(c.category)}» теряет в цене`,
    91→        ]
    92→    await db.marketEvent.create({
    93→      data: {
    94→        category: c.category, kind: up ? 'demand_up' : 'demand_down', magnitude: up ? magnitude : -magnitude,
    95→        headline: rnd(templates), body: up ? 'Продавцы поднимают цены, покупатели торопятся.' : 'Выгодные покупки — у тех, кто умеет ждать.',
    96→        expiresAt: new Date(Date.now() + randInt(2, 6) * 3_600_000),
    97→      },
    98→    })
    99→  }
   100→  // гаражные распродажи — отдельный цикл, примерно раз в 2 часа
   101→  if (tick % 8 === 0 && Math.random() < 0.12) {
   102→    await spawnGarageSale().catch((e) => console.error('[engine] garage:', e))
   103→  }
   104→  // ежедневное крупное событие
   105→  await dailySpecial()
   106→}

   197→// Гаражная распродажа: пачка discounted-объявлений в двух категориях + новость
   198→export async function spawnGarageSale() {
   199→  const active = await db.marketEvent.findFirst({
   200→    where: { kind: 'garage', expiresAt: { gt: new Date() } },
   201→  })
   202→  if (active) return // одна распродажа за раз
   203→  const cats = await db.marketIndex.findMany()
   204→  if (!cats.length) return
   205→  const cityNames = ['Черёмушках', 'Ярославском шоссе', 'Гавриловой-Яме', 'Лиговке', 'Бутове', 'Уралмаше']
   206→  const catA = rnd(cats)
   207→  let catB = rnd(cats)
   208→  if (catB.category === catA.category) catB = rnd(cats)
   209→  const botsAll = await db.user.findMany({ where: { isBot: true }, take: 100 })
   210→  if (!botsAll.length) return
   211→  let spawned = 0
   212→  for (const cat of [catA, catB]) {
   213→    const mult = cat.multiplier
   214→    const pool = CATALOG.filter((i) => i.category === cat.category && !i.key.startsWith('trash-'))
   215→    const n = Math.min(pool.length, randInt(4, 6))
   216→    for (let i = 0; i < n; i++) {
   217→      const item = rnd(pool)
   218→      const cond = conditionWeighted()
   219→      const full = item.basePrice * (CONDITION_MULT[cond] ?? 0.8) * mult
   220→      const price = Math.max(50, Math.round(full * (0.5 + Math.random() * 0.22)))
   221→      const bot = rnd(botsAll)
   222→      const created = await db.listing.create({
   223→        data: {
   224→          sellerId: bot.id, itemKey: item.key, title: item.title,
   225→          description: `Гаражная распродажа, всё за полцены. ${rnd(item.desc)}`,
   226→          category: item.category, condition: cond, price, baseValue: item.basePrice,
   227→          image: CATEGORY_IMAGE[item.category] ?? '/img/cat-electronics.jpg',
   228→          city: bot.city, createdAt: new Date(),
   229→        },
   230→      })
   231→      await onListingCreated(created)
   232→      spawned++
   233→    }
   234→  }
   235→  await db.marketEvent.create({
   236→    data: {
   237→      category: catA.category, kind: 'garage', magnitude: 0,
   238→      headline: `Гаражная распродажа в ${rnd(cityNames)}: ${spawned} лотов за полцены`,
   239→      body: `Соседи выносят всё: «${catLabel(catA.category)}» и «${catLabel(catB.category)}» почти даром. Успей, пока не разобрали.`,
   240→      expiresAt: new Date(Date.now() + 4 * 3_600_000),
   241→    },
   242→  })
   243→  cache.invalidate('feed')
   244→  // пуш всем живым игрокам
   245→  const players = await db.user.findMany({ where: { isBot: false }, take: 100 })
   246→  for (const p of players) {
   247→    await notifyUser(
   248→      p.id, 'market', 'Гаражная распродажа',
   249→      `${spawned} товаров за полцены в категориях «${catLabel(catA.category)}» и «${catLabel(catB.category)}». Рынок не будет ждать.`,
   250→    )
   251→  }
   252→}

   254→// ---------- БОТЫ ----------
   255→async function botsTick(tick: number) {
   256→  const botCount = await db.user.count({ where: { isBot: true } })
   257→  if (botCount === 0) return
   258→
   259→  // 1. Выставляют новые объявления
   260→  const active = await db.listing.count({ where: { status: 'active', seller: { isBot: true } } })
   261→  const freeActive = await db.listing.count({ where: { status: 'active', price: 0 } })
   262→  const want = Math.min(3, Math.max(0, 50 - active))
   263→  for (let i = 0; i < want; i++) {
   264→    const bot = await randomBot()
   265→    if (!bot) break
   266→    const item = weightedCatalogItem()
   267→    const cond = conditionWeighted()
   268→    const mult = await getCategoryMult(item.category)
   269→    const persona = personaOfId(bot.personaId)
   270→    const base = item.basePrice * (1 + (Math.random() - 0.5) * 2 * item.jitter)
   271→    let price = Math.round(base * (CONDITION_MULT[cond] ?? 0.8) * mult * (0.86 + persona.greed * 0.3 + Math.random() * 0.15))
   272→    const isJunk = item.key.startsWith('trash-') || item.basePrice < 500
   273→    if (isJunk && freeActive < 3 && Math.random() < 0.4) {
   274→      price = 0
   275→    }
   276→    if (!isJunk && price < 100) price = 100
   277→    const created = await db.listing.create({
   278→      data: {
   279→        sellerId: bot.id, itemKey: item.key, title: item.title,
   280→        description: rnd(item.desc), category: item.category, condition: cond,
   281→        price, baseValue: item.basePrice, image: CATEGORY_IMAGE[item.category] ?? '/img/cat-electronics.jpg',
   282→        city: bot.city,
   283→        createdAt: new Date(Date.now() - randInt(0, 90) * 60_000),
   284→      },
   285→    })
   286→    await onListingCreated(created)
   287→  }
   288→
   289→  // 2. Конкуренция: продавцы одного товара сбивают цену друг другу
   290→  if (tick % 4 === 0) {
   291→    const actives = await db.listing.findMany({
   292→      where: { status: 'active' },
   293→      include: { seller: true },
   294→      orderBy: { createdAt: 'desc' },
   295→      take: 300,
   296→    })
   297→    const groups = new Map<string, typeof actives>()
   298→    for (const l of actives) {
   299→      const arr = groups.get(l.itemKey) ?? []
   300→      arr.push(l)
   301→      groups.set(l.itemKey, arr)
   302→    }
   303→    for (const [, list] of groups) {
   304→      if (list.length < 2) continue
   305→      const sorted = [...list].sort((a, b) => a.price - b.price)
   306→      const cheapest = sorted[0]
   307→      const overpriced = sorted[sorted.length - 1]
   308→      if (overpriced.price <= cheapest.price) continue
   309→      if (overpriced.sellerId === cheapest.sellerId) continue
   310→      if (Date.now() - overpriced.createdAt.getTime() < 10 * 60_000) continue
   311→      if (Math.random() > 0.5) continue
   312→      const newPrice = Math.round(overpriced.price * (1 - 0.04 - Math.random() * 0.05))
   313→      if (newPrice <= cheapest.price || newPrice < 50) continue
   314→      if (overpriced.seller.isBot) {
   315→        const oldPrice = overpriced.price
   316→        await db.listing.update({ where: { id: overpriced.id }, data: { price: newPrice } })
   317→        await notifyPriceDrop({ id: overpriced.id, sellerId: overpriced.sellerId, title: overpriced.title, price: newPrice }, oldPrice)
   318→      } else {
   319→        // игроку — только уведомление, что конкурент давит
   320→        const rivalBot = sorted.find((l) => l.seller.isBot && l.sellerId !== overpriced.sellerId)
   321→        if (rivalBot && Math.random() < 0.6) {
   322→          const rp = Math.round(rivalBot.price * (1 - 0.04 - Math.random() * 0.05))
   323→          if (rp > cheapest.price && rp >= 50) {
   324→            const rivalOld = rivalBot.price
   325→            await db.listing.update({ where: { id: rivalBot.id }, data: { price: rp } })
   326→            await notifyPriceDrop({ id: rivalBot.id, sellerId: rivalBot.sellerId, title: rivalBot.title, price: rp }, rivalOld)
   327→          }
   328→          await notifyUser(
   329→            overpriced.sellerId, 'market', 'Конкурент сбивает цену',
   330→            `Такой же «${overpriced.title}» уже продают за ${fmtMoney(Math.min(rp, cheapest.price))}. Покупатели уходят к нему — подумайте о цене.`,
   331→          )
   332→        }
   333→      }
   334→    }
   335→  }
   336→
   337→  // 3. Скидки на залежалый товар
   338→  if (tick % 6 === 0) {
   339→    const stale = await db.listing.findMany({
   340→      where: {
   341→        status: 'active', createdAt: { lt: new Date(Date.now() - 2 * 3_600_000) },
   342→        price: { gt: 0 }, seller: { isBot: true },
   343→      },
   344→      take: 6, orderBy: { createdAt: 'asc' },
   345→    })
   346→    for (const l of stale) {
   347→      if (Math.random() < 0.5) continue
   348→      const newPrice = Math.round(l.price * 0.95)
   349→      const floor = Math.round(l.baseValue * 0.35)
   350→      if (newPrice < floor) continue
   351→      const oldPrice = l.price
   352→      await db.listing.update({ where: { id: l.id }, data: { price: newPrice } })
   353→      await notifyPriceDrop({ id: l.id, sellerId: l.sellerId, title: l.title, price: newPrice }, oldPrice)
   354→    }
   355→  }
   356→
   357→  // 4. Боты скупают выгодное
   358→  if (tick % 3 === 0) {
   359→    const cats = await db.marketIndex.findMany()
   360→    const allActive = await db.listing.findMany({
   361→      where: { status: 'active', price: { gt: 0 } },
   362→      include: { seller: true },
   363→      orderBy: { createdAt: 'asc' }, take: 60,
   364→    })
   365→    for (const l of allActive) {
   366→      if (Math.random() < 0.85) continue
   367→      const idxMult = (cats.find((c) => c.category === l.category)?.multiplier ?? 1) * (CONDITION_MULT[l.condition] ?? 0.8)
   368→      const est = l.baseValue * idxMult
   369→      if (l.price > est * 0.82) continue
   370→      const bot = await randomBot()
   371→      if (!bot || bot.id === l.sellerId) continue
   372→      await completeSale({ listingId: l.id, buyer: bot, price: l.price, via: 'engine' })
   373→      break // максимум одна движка-покупка за тик
   374→    }
   375→  }
   376→
   377→  // 5. Боты забирают халяву
   378→  const freebies = await db.listing.findMany({
   379→    where: { status: 'active', price: 0 },
   380→    include: { seller: true }, take: 3,
   381→  })
   382→  for (const f of freebies) {
   383→    if (Math.random() < 0.6) {
   384→      const bot = await randomBot()
   385→      if (bot && bot.id !== f.sellerId) {
   386→        await completeSale({ listingId: f.id, buyer: bot, price: 0, via: 'engine' })
   387→      }
   388→    }
   389→  }
   390→
   391→  // 6. Боты стучатся к игрокам в чат
   392→  if (tick % 2 === 0) {
   393→    const playerListings = await db.listing.findMany({
   394→      where: { status: 'active', seller: { isBot: false }, createdAt: { lt: new Date(Date.now() - 3 * 60_000) } },
   395→      take: 10, orderBy: { createdAt: 'desc' },
   396→    })
   397→    for (const pl of playerListings) {
   398→      if (Math.random() > 0.22) continue
   399→      const existing = await db.chat.findMany({ where: { listingId: pl.id, sellerId: pl.sellerId } })
   400→      const botChats = existing.filter((c) => c.buyerId !== pl.sellerId)
   401→      if (botChats.length >= 2) continue
   402→      const bot = await randomBot()
   403→      if (!bot) continue
   404→      const meta = { botRole: 'buyer' as const, botLimit: Math.round(pl.price * (1 - 0.05 - Math.random() * 0.12)), rounds: 0 }
   405→      const chat = await db.chat.create({
   406→        data: { listingId: pl.id, buyerId: bot.id, sellerId: pl.sellerId, meta: JSON.stringify(meta) },
   407→      })
   408→      const opener = botOpener(chat, pl, bot)
   409→      const m = await db.message.create({
   410→        data: {
   411→          chatId: chat.id, senderType: 'bot', senderId: bot.id, senderName: bot.displayName,
   412→          kind: 'text', text: opener.text,
   413→        },
   414→      })
   415→      await db.notification.create({
   416→        data: { userId: pl.sellerId, kind: 'message', title: bot.displayName, body: m.text.slice(0, 80) },
   417→      })
   418→      await emitTo(`user:${pl.sellerId}`, 'notify', {

   420→      })

   427→async function auctionTick(tick: number) {
   428→  const activeLots = await db.auctionLot.findMany({ where: { status: 'active' } })
   429→
   430→  // создать лоты
   431→  const valuable = CATALOG.filter((i) => i.basePrice >= 12_000)
   432→  if (activeLots.length < 3 && Math.random() < 0.18 && valuable.length) {
   433→    const item = rnd(valuable)
   434→    const cond = Math.random() < 0.5 ? 'excellent' : 'good'
   435→    const mult = await getCategoryMult(item.category)
   436→    const est = Math.round(item.basePrice * (CONDITION_MULT[cond] ?? 1) * mult)
   437→    await db.auctionLot.create({
   438→      data: {
   439→        itemKey: item.key, title: item.title, image: CATEGORY_IMAGE[item.category] ?? '/img/cat-electronics.jpg',
   440→        category: item.category, condition: cond, baseValue: item.basePrice,
   441→        startPrice: Math.round(est * (0.5 + Math.random() * 0.2)),
   442→        endsAt: new Date(Date.now() + randInt(15, 45) * 60_000),
   443→      },
   444→    })
   445→  }
   446→
   447→  // ставки ботов
   448→  if (tick % 2 === 0) {
   449→    for (const lot of activeLots) {
   450→      if (lot.endsAt.getTime() - Date.now() < 30_000) continue
   451→      if (Math.random() > 0.3) continue
   452→      const step = auctionStep(lot.startPrice)
   453→      const mult = await getCategoryMult(lot.category)
   454→      const est = Math.round(lot.baseValue * (CONDITION_MULT[lot.condition] ?? 1) * mult)
   455→      const privateLimit = Math.round(est * (0.85 + Math.random() * 0.5))
   456→      const nextBid = (lot.currentBid ?? lot.startPrice) + step
   457→      if (nextBid > privateLimit) continue
   458→      // перебить
   459→      const prevBidderId = lot.currentBidderId
   460→      const bot = await randomBot()
   461→      if (!bot) continue
   462→      if (bot.id === prevBidderId) continue
   463→      await db.auctionBid.create({
   464→        data: { lotId: lot.id, userId: bot.id, userName: bot.displayName, amount: nextBid },
   465→      })
   466→      await db.auctionLot.update({
   467→        where: { id: lot.id },
   468→        data: { currentBid: nextBid, currentBidderId: bot.id, currentBidderName: bot.displayName, bidCount: { increment: 1 } },
   469→      })
   470→      // вернуть деньги игроку, которого перебили
   471→      if (prevBidderId) {
   472→        const prev = await db.user.findUnique({ where: { id: prevBidderId } })
   473→        if (prev && !prev.isBot && lot.currentBid) {
   474→          await db.user.update({ where: { id: prev.id }, data: { balance: { increment: lot.currentBid } } })
   475→          await notifyUser(prev.id, 'market', 'Вас перебили на аукционе', `Лот «${lot.title}» уходит за ${fmtMoney(nextBid)}. Ставка возвращена на счёт.`)
   476→        }
   477→      }
   478→      await emitTo('global', 'auction:update', { lotId: lot.id })
   479→    }
   480→  }
   481→
   482→  // завершение
   483→  const due = await db.auctionLot.findMany({
   484→    where: { status: 'active', endsAt: { lte: new Date() } },
   485→  })
   486→  for (const lot of due) {
   487→    if (!lot.currentBidderId || !lot.currentBid) {
   488→      await db.auctionLot.update({ where: { id: lot.id }, data: { status: 'cancelled', finishedAt: new Date() } })
   489→      continue
   490→    }
   491→    const winner = await db.user.findUnique({ where: { id: lot.currentBidderId } })
   492→    if (!winner) continue
   493→    if (!winner.isBot) {
   494→      await db.transaction.create({
   495→        data: { userId: winner.id, type: 'purchase', amount: -lot.currentBid, note: `Аукцион: ${lot.title}` },
   496→      })
   497→      await notifyUser(winner.id, 'deal', 'Вы выиграли аукцион', `«${lot.title}» за ${fmtMoney(lot.currentBid)} уже в инвентаре`)
   498→    } else {
   499→      await db.user.update({ where: { id: winner.id }, data: { balance: { decrement: lot.currentBid } } })
   500→    }
   501→    const item = await db.item.create({
   502→      data: {
   503→        ownerId: winner.id, itemKey: lot.itemKey, title: lot.title, category: lot.category,
   504→        condition: lot.condition, image: lot.image, baseValue: lot.baseValue,
   505→        purchasePrice: lot.currentBid, fromUserId: null,
   506→      },
   507→    })
   508→    await db.auctionLot.update({ where: { id: lot.id }, data: { status: 'finished', finishedAt: new Date() } })
   509→    await emitTo('global', 'auction:update', { lotId: lot.id, finished: true })
   510→  }
   511→}

   514→async function financeTick() {
   515→  // проценты по вкладам
   516→  const users = await db.user.findMany({
   517→    where: { deposit: { gt: 0 }, depositAt: { lt: new Date(Date.now() - 3_600_000) } },
   518→    take: 50,
   519→  })
   520→  for (const u of users) {
   521→    if (!u.depositAt) continue
   522→    const hours = Math.floor((Date.now() - u.depositAt.getTime()) / 3_600_000)
   523→    if (hours < 1) continue
   524→    const interest = Math.round(u.deposit * 0.001 * hours)
   525→    if (interest <= 0) continue
   526→    await db.user.update({
   527→      where: { id: u.id },
   528→      data: { deposit: { increment: interest }, depositAt: new Date(u.depositAt.getTime() + hours * 3_600_000) },
   529→    })
   530→    if (!u.isBot) {
   531→      await db.transaction.create({ data: { userId: u.id, type: 'interest', amount: interest, note: `Проценты по вкладу (${hours} ч.)` } })
   532→    }
   533→  }
   534→  // пеня по налогам
   535→  const debtors = await db.user.findMany({
   536→    where: { isBot: false, taxDebt: { gt: 0 }, taxPenaltyAt: { lt: new Date(Date.now() - 86_400_000) } },
   537→  })
   538→  for (const u of debtors) {
   539→    const penalty = Math.round(u.taxDebt * 0.1)
   540→    await db.user.update({
   541→      where: { id: u.id },
   542→      data: { taxDebt: { increment: penalty }, taxPenaltyAt: new Date() },
   543→    })
   544→    await db.taxBill.create({
   545→      data: { userId: u.id, amount: penalty, reason: 'Пеня 10% за просрочку налога', dueAt: new Date(Date.now() + 86_400_000) },
   546→    })
   547→    await notifyUser(u.id, 'tax', 'Пеня от ФНС', `Начислена пеня ${fmtMoney(penalty)}. Задолженность растёт каждый день.`)
   548→  }
   549→  // просроченные кредиты
   550→  const overdue = await db.loan.findMany({ where: { status: 'active', dueAt: { lt: new Date() } } })
   551→  for (const l of overdue) {
   552→    await db.loan.update({ where: { id: l.id }, data: { status: 'overdue' } })
   553→    const owner = await db.user.findUnique({ where: { id: l.userId } })
   554→    if (owner && !owner.isBot) {
   555→      const newScore = Math.max(300, owner.creditScore - 80)
   556→      await db.user.update({ where: { id: owner.id }, data: { creditScore: newScore } })
   557→      await notifyUser(l.userId, 'system', 'Кредит просрочен', `Банк ждёт погашения. Кредитный рейтинг упал до ${newScore} — лимит срезан, ставка выросла. Покупки ограничены при долге свыше 30 000 ₽.`)
   558→    } else {
   559→      await notifyUser(l.userId, 'system', 'Кредит просрочен', 'Банк ждёт погашения. Покупки ограничены при долге свыше 30 000 ₽.')
   560→    }
   561→  }
   562→}

   593→export async function tickAll() {
   594→  if (!g.__avitoEngine) return
   595→  g.__avitoEngine.tick++
   596→  const tick = g.__avitoEngine.tick
   597→  try {
   598→    await marketTick(tick)
   599→    await botsTick(tick)
   600→    await auctionTick(tick)
   601→    await deliveryTick()
   602→    await repairTick()
   603→    if (tick % 4 === 0) await financeTick()
   604→    await presenceTick()
   605→    if (tick % 8 === 0) await ensureDailyQuestsAll()
   606→  } catch (e) {
   607→    console.error('[engine] tick error:', e)
   608→  }
   609→}
