    17→    17→// Тик раз в секунду через useSyncExternalStore — таймеры лотов живые, без setState внутри эффектов
    18→    18→function useTick(intervalMs = 1000): number {
    19→    19→  return useSyncExternalStore(
    20→    20→    (cb) => {
    21→    21→      const id = setInterval(cb, intervalMs)
    22→    22→      return () => clearInterval(id)
    23→    23→    },
    24→    24→    () => Math.floor(Date.now() / intervalMs),
    25→    25→    () => 0,
    26→    26→  )
    27→    27→}

    29→    29→function bidStep(lot: AuctionLotDTO): number {
    30→    30→  return Math.max(100, Math.round((lot.currentBid ?? lot.startPrice) * 0.02))
    31→    31→}

    33→    33→function minBid(lot: AuctionLotDTO): number {
    34→    34→  return (lot.currentBid ?? lot.startPrice) + bidStep(lot)

    37→    37→function fmtTimer(ms: number): string {
    38→    38→  const s = Math.max(0, Math.floor(ms / 1000))
    39→    39→  const h = Math.floor(s / 3600)
    40→    40→  const m = Math.floor((s % 3600) / 60)
    41→    41→  const ss = s % 60
    42→    42→  if (h > 0) return `${h}ч ${m}м ${String(ss).padStart(2, '0')}с`
    43→    43→  if (m > 0) return `${m}м ${String(ss).padStart(2, '0')}с`
    44→    44→  return `${ss} с`
    45→    45→}

    47→    47→function plural(n: number, one: string, few: string, many: string): string {
    48→    48→  const m10 = n % 10
    49→    49→  const m100 = n % 100
    50→    50→  if (m10 === 1 && m100 !== 11) return one
    51→    51→  if (m10 >= 2 && m10 <= 4 && (m100 < 10 || m100 >= 20)) return few
    52→    52→  return many
    53→    53→}
